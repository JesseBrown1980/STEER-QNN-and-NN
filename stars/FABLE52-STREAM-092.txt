typedef struct node {
    int key;
    struct node *left;
    struct node *right;
} node_t;

static node_t *insert(node_t *root, int key) {
    if (root == NULL) {
        node_t *n = malloc(sizeof(node_t));
        n->key = key;
        n->left = n->right = NULL;
        return n;
    }
    if (key < root->key) root->left = insert(root->left, key);
    else if (key > root->key) root->right = insert(root->right, key);
    return root;
}

}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Trit { Neg = -1, Zero = 0, Pos = 1 }

impl Trit {
    pub fn from_i8(v: i8) -> Option<Self> {
        match v {
            -1 => Some(Trit::Neg),
            0 => Some(Trit::Zero),
            1 => Some(Trit::Pos),
            _ => None,
        }
    }
}

pub fn to_balanced(mut n: i64) -> Vec<Trit> {
    let mut out = Vec::new();
    while n != 0 {
        let r = n.rem_euclid(3);
        let (t, carry) = if r == 2 { (Trit::Neg, 1) } else if r == 1 { (Trit::Pos, 0) } else { (Trit::Zero, 0) };
        out.push(t);
        n = n.div_euclid(3) + carry;
    }
    out
}

class Reader:
    def __init__(self, order=2):
        self.order = order
        self.seen = defaultdict(int)
        self.hit = defaultdict(int)

    def cost(self, ctx, sym):
        s, h = self.seen[ctx], self.hit[(ctx, sym)]
        self.seen[ctx] += 1
        self.hit[(ctx, sym)] += 1
        return log2_256(s + 256) - log2_256(h + 1)

def sha256_file(path, block=1 << 20):
    h = hashlib.sha256()
    with open(path, "rb") as f:
        while True:
            b = f.read(block)
            if not b:
                break
            h.update(b)
    return h.hexdigest()

function debounce(fn, wait) {
  let timer = null;
  return function (...args) {
    clearTimeout(timer);
    timer = setTimeout(() => fn.apply(this, args), wait);
  };
}
