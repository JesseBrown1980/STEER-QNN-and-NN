defmodule Star do
  @n 64
  def project(body) when is_binary(body) do
    body
    |> :binary.bin_to_list()
    |> Enum.map(&div(&1, 4))
    |> Enum.chunk_every(3, 3, :discard)
    |> Enum.frequencies()
  end

  def emit(cube, u, v) do
    cube
    |> Enum.filter(fn {[a, b, _], _} -> a == u and b == v end)
    |> Enum.max_by(fn {[_, _, w], c} -> {c, -w} end, fn -> nil end)
  end
end
