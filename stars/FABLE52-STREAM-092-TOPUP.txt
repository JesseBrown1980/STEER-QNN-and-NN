type Row struct {
	Tag    string
	Fields map[string]string
}

func parse(line string) Row {
	parts := strings.Split(line, "|")
	r := Row{Tag: parts[0], Fields: make(map[string]string)}
	for _, p := range parts[1:] {
		if i := strings.IndexByte(p, '='); i >= 0 {
			r.Fields[p[:i]] = p[i+1:]
		}
	}
	return r
}

	h := sha256.New()
	counts := map[string]int{}
	sc := bufio.NewScanner(f)
	for sc.Scan() {
		line := sc.Text()
		h.Write([]byte(line + "\n"))
		counts[parse(line).Tag]++
	}
	tags := make([]string, 0, len(counts))
	for t := range counts {
		tags = append(tags, t)
	}
	sort.Strings(tags)
	for _, t := range tags {
		fmt.Printf("COUNT|tag=%s|rows=%d|json=0\n", t, counts[t])
	}
	fmt.Println("SHA|" + hex.EncodeToString(h.Sum(nil)))
}

    public Matrix(int n) {
        this.n = n;
        this.a = new long[n][n];
    }

    public Matrix multiply(Matrix other) {
        Matrix out = new Matrix(n);
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < n; j++) {
                long s = 0;
                for (int k = 0; k < n; k++) {
                    s = (s + a[i][k] * other.a[k][j]) % MOD;
                }
                out.a[i][j] = s;

class Stack {
public:
    void push(const T& value) { items_.push_back(value); }
    T pop() {
        if (items_.empty()) throw std::out_of_range("empty stack");
        T v = items_.back();
        items_.pop_back();
        return v;
    }
    bool empty() const { return items_.empty(); }
    std::size_t size() const { return items_.size(); }
private:
    std::vector<T> items_;
};
